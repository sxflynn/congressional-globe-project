def canonical(value):
    return json.dumps(value, sort_keys=True, ensure_ascii=False, separators=(',', ':'))

def sha(data):
    return hashlib.sha256(data).hexdigest()

def save(path, value):
    path.write_text(json.dumps(value, indent=2, ensure_ascii=False) + '\n')

manifest = json.loads((OLD/'installed-model-manifest.json').read_text())
layer = next(x for x in manifest['layers'] if x.get('name') == 'tokenizer.json')

assert sha(resource.read_bytes()) == layer['digest'].split(':')[1]
tok = Tokenizer.from_file(str(resource))

def measure(text):
    return {'utf8_bytes': len(text.encode()), 'codepoints': len(text),
            'tokens': len(tok.encode(text, add_special_tokens=False).ids),
            'sha256': sha(text.encode())}

# A reversible column encoding of diagnostic rows, leaving historical strings verbatim.
# This proposal has no production decoder and cannot pass the current input contract.
def compact_provider(provider):
    value = copy.deepcopy(provider)
    signals = value['diagnostics'].pop('signals')
    columns = sorted(set().union(*(s.keys() for s in signals)))
    reasons = sorted({s['route_reason'] for s in signals})
    value['diagnostics']['signal_table'] = {
        'columns': columns, 'route_reasons': reasons,
        'rows': [[reasons.index(s[k]) if k == 'route_reason' else s[k] for k in columns] for s in signals]}
    value['contract'] = 'ocr-productive-repair-input-v3-column-proposal'
    return value

def restore_provider(value):
    value = copy.deepcopy(value)
    table = value['diagnostics'].pop('signal_table')
    value['diagnostics']['signals'] = [
        {k: table['route_reasons'][v] if k == 'route_reason' else v for k, v in zip(table['columns'], row, strict=True)}
        for row in table['rows']]
    value['contract'] = 'ocr-productive-repair-input-v2'
    return value

In the route_reason column each integer selects route_reasons by zero-based index. Expand each row using those columns and reasons. All original fields, flags, quotes and offsets retain their meanings. No target or evidence is omitted. The reply contract is unchanged.'
units = json.loads((INPUT/'units.json').read_text())
for meta in units:
    unit = meta['unit_id']
    if meta['status'] == 'missing_source_image':
        rows.append({'unit_id': unit, 'status': 'missing_source_image', 'measurements': None})
        continue
    folder = OUT/unit
    folder.mkdir(exist_ok=True)
    src = INPUT/unit
    projection = json.loads((src/'projection.json').read_text())
    raw_request = (src/'request.json').read_bytes()
    assert sha(raw_request) == meta['request_sha256'] == projection['body_sha256']
    assert raw_request == projection['body_utf8'].encode()
    request = json.loads(raw_request)
    host = json.loads((src/'host.json').read_text())
    schema = request['format']
    assert sha(canonical(schema).encode()) == meta['schema_sha256']
    validator = Draft202012Validator(schema)
    user = request['messages'][1]['content']
    prompt = request['messages'][0]['content']
    provider = json.loads(user)
    targets = [p for p in host['payload']['paragraphs'] if p['role'] == 'target']
    image = base64.b64decode(request['messages'][1]['images'][0], validate=True)
    assert sha(image) == meta['image_sha256']

    base = {'contract':'ocr-productive-repair-reply-v2','unit_id':unit,'role':role,

            'self_check':[],'findings':[],'patches':[],'target_readings':[]}
    for p in targets:
        alias = p['alias']
        base['inspections'].append({'paragraph_alias':alias,'image_alias':'I1',
            'completeness':'not_inspected','image_suitability':'unclear','reason':'Not inspected.'})
        base['self_check'].append({'paragraph_alias':alias,'coverage':'not_inspected',
            'checked_content':[],'remaining_concerns':[]})
        if not audit:
            base['target_readings'].append({'paragraph_alias':alias,'image_alias':'I1',
                'reading_status':'withheld','proposed_text':None,'uncertainties':[], 'reason':'Not inspected.'})
    full = copy.deepcopy(base)
    full['status'], full['reason'] = 'no_finding', None
    for i, p in enumerate(targets):
        full['inspections'][i].update(completeness='complete', image_suitability='clear', reason=None)
        full['self_check'][i].update(coverage='complete', checked_content=checks)
        if not audit:
            full['target_readings'][i].update(reading_status='unchanged',proposed_text=p['raw']['text'],reason=None)
    variants = {'small_incomplete':base,'full_scope_raw_copy_illustration':full}
    if audit:
        concerned = copy.deepcopy(full)
        concerned.update(status='unresolved', reason='Illustrative concerns remain.')
        for i, p in enumerate(targets, 1):
            concerned['self_check'][i-1]['remaining_concerns'] = ['Illustrative unresolved concern.']
            concerned['findings'].append({'finding_id':f'F{i}',
                'anchor':{'paragraph_alias':p['alias'],'scope':'whole_paragraph','raw_quote':None},
                'affected_layer':'saved_proposal','affected_quote':None,
                'category':'unsupported_saved_proposal','resolution':'unresolved',
                'proposed_reading':None,'image_alias':'I1','basis':'image_unclear',
                'explanation':'Illustrative unresolved concern.'})
        variants['one_unresolved_concern_per_target_illustration'] = concerned
    if not audit:
        edited = copy.deepcopy(full)
        edits = case_proposals[unit]['all_edits_from_raw'] if unit in case_proposals else [
            {'raw_quote': e['expected'], 'replacement':e['replacement'], 'start':e['raw_start'],'end':e['raw_end']}
            for e in targets[0]['static']['edits']]
        raw = targets[0]['raw']['text']
        proposed = raw
        for e in sorted(edits, key=lambda e:e['start'], reverse=True):
            assert proposed[e['start']:e['end']] == e['raw_quote']
            proposed = proposed[:e['start']] + e['replacement'] + proposed[e['end']:]
        for i,e in enumerate(edits,1):
            fid = f'F{i}'
            edited['findings'].append({'finding_id':fid,'anchor':{'paragraph_alias':'P1','scope':'quote','raw_quote':e['raw_quote']},'affected_layer':'raw','affected_quote':e['raw_quote'],'category':'other_damage','resolution':'proposed','proposed_reading':e['replacement'],'image_alias':'I1','basis':'visible_print','explanation':'Illustrative saved edit.'})
            edited['patches'].append({'finding_id':fid,'paragraph_alias':'P1','raw_quote':e['raw_quote'],'replacement':e['replacement'],'image_alias':'I1','reason':'Illustrative saved edit.'})
        if edits:
            edited['status']='findings'
            edited['target_readings'][0].update(reading_status='proposed',proposed_text=proposed)
        variants['saved_edits_full_reading_illustration']=edited
    for name, reply in variants.items():

        validator.validate(reply)

            'named_feedback':'raw-copy illustration leaves known feedback unaddressed' if unit in case_proposals and 'raw_copy' in name else 'not a new outcome'})

        'image_transport':{'decoded_bytes':len(image),'sha256':sha(image),'base64_ascii_bytes':len(request['messages'][1]['images'][0]),'native_image_tokens':None},
        'options':request['options'],'native_fit':None,'dispatch_allowed':False})
