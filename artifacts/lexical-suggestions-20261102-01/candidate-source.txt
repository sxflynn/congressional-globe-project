def distance(a,b):
 row=list(range(len(b)+1))
 for i,left in enumerate(a,1):
  new=[i]
  for j,right in enumerate(b,1):new.append(min(new[-1]+1,row[j]+1,row[j-1]+(left!=right)))
  row=new
 return row[-1]
def rank(text):

 score=lambda candidate:(distance(normalized,candidate),-len(os.path.commonprefix([normalized,candidate])),abs(len(candidate)-len(normalized)),candidate)
 top=heapq.nsmallest(5,pool,key=score)
 return dict(input=text,normalized_letters=normalized,candidates_scored=len(pool),elapsed_seconds=time.perf_counter()-started,suggestions=[dict(form=x,letter_distance=distance(normalized,x),shared_prefix=len(os.path.commonprefix([normalized,x])),in_word_list=x in word_forms,in_name_list=x in name_forms) for x in top])
