import json,subprocess,time
from pathlib import Path
root=Path('artifacts/s2-consistency-20261102-01');pids=set();pgids=set()
for i in (1,2,3):
 b=root/f'repeat-{i}/run';op=json.loads((b/'operator-process.json').read_text());pids.add(op['pid'])
 fields=op['process'].splitlines()[-1].split(None,2);pids.add(int(fields[1]))
 for p in (b/'spool').glob('*/receipt.json'):
  r=json.loads(p.read_text());assert r['stop_observation']=='stopped';pids.update(x['pid'] for x in r['known_processes']);pgids.add(r['owned_pgid'])
ps=subprocess.run(['ps','-axo','pid,ppid,pgid,lstart,etime,command'],check=True,capture_output=True,text=True).stdout
rows=[]
for line in ps.splitlines()[1:]:
 f=line.split(None,3)
 if len(f)>3 and (int(f[0]) in pids or int(f[2]) in pgids):rows.append(line.strip())
assert not rows,'Owned identity found; inspect before any completion claim.'
(root/'terminal-processes.json').write_text(json.dumps(dict(checked_at_epoch=time.time(),owned_pids=sorted(pids),owned_pgids=sorted(pgids),matching_rows=rows,observation='No owned operator/uv-parent/server/child identities present; historical receipts unchanged.'),indent=2)+'\n')
print('All three positive shutdown receipts and terminal owned-identity absence verified.')
